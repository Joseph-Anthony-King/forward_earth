### Planetary Union

# Democratic Generic 1
empire_name_format = {
	random_weight = {
		factor = 0
		modifier = {
			add = 1
			has_government = "gov_planetary_union"
			is_pirate = no
			is_primitive = no
			is_nomadic = no
			NOT = { is_country_type = fallen_empire }
			NOT = { is_country_type = awakened_fallen_empire }
		}
	}
	format = "{AofB{<democratic_gen> [This.GetCapitalSystemNameOrRandom]}}"
	noun = "{[This.GetCapitalSystemNameOrRandom]}"
	prefix_format = "{AofBpfx{[This.GetCapitalSystemNameOrRandom] <democratic_gen>}}"
}
